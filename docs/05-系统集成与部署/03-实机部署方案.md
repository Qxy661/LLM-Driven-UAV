## 2.3 LLM API 调用的错误处理与重试

在实机部署中，LLM API 调用可能因网络波动、速率限制或服务暂时不可用而失败。以下是一个带重试逻辑的健壮调用封装：

```python
import time
import logging
from openai import OpenAI, RateLimitError, APITimeoutError, APIConnectionError

logger = logging.getLogger(__name__)

class RobustLLMClient:
    """带重试和错误处理的 LLM 客户端，适用于无人机实机部署"""

    def __init__(self, api_key: str, model: str = "gpt-4o",
                 max_retries: int = 3, base_delay: float = 1.0,
                 timeout: float = 10.0, fallback_model: str = "gpt-4o-mini"):
        self.client = OpenAI(api_key=api_key)
        self.model = model
        self.max_retries = max_retries
        self.base_delay = base_delay
        self.timeout = timeout
        self.fallback_model = fallback_model

    def generate(self, prompt: str, system_prompt: str = "", max_tokens: int = 512) -> str:
        """调用 LLM，带指数退避重试和降级策略"""
        messages = []
        if system_prompt:
            messages.append({"role": "system", "content": system_prompt})
        messages.append({"role": "user", "content": prompt})

        current_model = self.model
        last_exception = None

        for attempt in range(self.max_retries + 1):
            try:
                response = self.client.chat.completions.create(
                    model=current_model,
                    messages=messages,
                    max_tokens=max_tokens,
                    timeout=self.timeout
                )
                result = response.choices[0].message.content.strip()
                logger.info(f"LLM 调用成功 (模型={current_model}, 尝试={attempt + 1})")
                return result

            except RateLimitError as e:
                # 速率限制：等待后重试
                last_exception = e
                wait_time = self.base_delay * (2 ** attempt)
                logger.warning(f"速率限制，等待 {wait_time:.1f}s 后重试 ({attempt + 1}/{self.max_retries})")
                time.sleep(wait_time)

            except APITimeoutError as e:
                # 超时：缩短 prompt 后重试
                last_exception = e
                logger.warning(f"API 超时 ({attempt + 1}/{self.max_retries}): {e}")
                if attempt >= 1:
                    # 第二次起缩短输入
                    messages[-1]["content"] = messages[-1]["content"][:500]
                time.sleep(self.base_delay)

            except APIConnectionError as e:
                # 网络问题：等待更久后重试
                last_exception = e
                wait_time = self.base_delay * (2 ** attempt) * 2
                logger.warning(f"网络连接失败，等待 {wait_time:.1f}s ({attempt + 1}/{self.max_retries})")
                time.sleep(wait_time)

            except Exception as e:
                # 未知错误：记录后尝试降级
                last_exception = e
                logger.error(f"未知错误: {type(e).__name__}: {e}")
                break

        # 所有重试失败 → 尝试降级到更小/更快的模型
        if current_model != self.fallback_model:
            logger.warning(f"主模型 {current_model} 失败，降级到 {self.fallback_model}")
            current_model = self.fallback_model
            try:
                response = self.client.chat.completions.create(
                    model=self.fallback_model,
                    messages=messages,
                    max_tokens=max_tokens,
                    timeout=self.timeout
                )
                return response.choices[0].message.content.strip()
            except Exception as fallback_e:
                logger.error(f"降级模型也失败: {fallback_e}")

        # 完全失败 → 返回安全默认响应
        logger.error(f"LLM 调用彻底失败: {last_exception}")
        return "[LLM 不可用] 无法获取响应，请执行预设安全动作。"


# ---- 使用示例 ----
llm = RobustLLMClient(
    api_key="your-api-key",
    model="gpt-4o",
    max_retries=3,
    base_delay=1.0,
    timeout=10.0,
    fallback_model="gpt-4o-mini"
)

try:
    result = llm.generate(
        prompt="将以下无人机指令转为航点序列: 巡逻A区域，高度50米",
        system_prompt="你是无人机任务规划助手，输出 JSON 格式航点列表。"
    )
    print(result)
except Exception as e:
    # 最终兜底：切换到预设任务
    print(f"LLM 不可用，执行预设巡逻任务: {e}")
```

```python
class RealTimeOptimizer:
    """实时性优化器"""

    def __init__(self):
        self.task_priorities = {
            "emergency": 0,    # 最高优先级
            "safety": 1,
            "control": 2,
            "planning": 3,
            "analysis": 4,     # 最低优先级
        }

    def schedule_task(self, task, priority):
        """调度任务"""
        if priority == "emergency":
            # 紧急任务立即执行
            return self._execute_immediately(task)
        elif priority == "safety":
            # 安全任务高优先级
            return self._execute_high_priority(task)
        else:
            # 其他任务排队执行
            return self._queue_task(task, priority)
```

```python
class PowerManager:
    """功耗管理器"""

    def __init__(self):
        self.power_budget = {
            "compute": 25,  # W
            "motors": 200,  # W
            "sensors": 5,   # W
            "comm": 3,      # W
        }

    def optimize_power(self, battery_level):
        """根据电量优化功耗"""
        if battery_level < 30:

            self._reduce_compute_power()
            # 降低 LLM 推理频率
```

### 5.2 通信管理

```python
class CommunicationManager:
    """通信管理器"""

    def __init__(self):
        self.links = {
            "wifi": {"status": "unknown", "bandwidth": 0},
            "4g": {"status": "unknown", "bandwidth": 0},
        }
        self.active_link = None

    def select_best_link(self):
        """选择最佳通信链路"""
        available = [
            (name, info) for name, info in self.links.items()
            if info["status"] == "connected"
        ]

        if not available:
            return None

        # 选择带宽最高的
        best = max(available, key=lambda x: x[1]["bandwidth"])
        self.active_link = best[0]
        return best[0]

    def send_with_priority(self, data, priority):
        """按优先级发送数据"""
        if priority == "critical":
            # 关键数据立即发送
            return self._send_immediately(data)
        elif priority == "high":
            # 高优先级数据
            return self._send_queued(data, position="front")
        else:
            # 普通数据排队
            return self._send_queued(data, position="back")
```

<summary>参考答案</summary>

**1.** 策略：(1) 使用量化模型减少计算量；(2) 根据任务阶段调整推理频率（巡航时少推理，执行任务时多推理）；(3) 在电量充足时使用云端推理，电量不足时切换到边缘；(4) 动态调整模型大小（电量高用7B，电量低用3B）。

**2.** Orin Nano 适合：简单指令理解、基础场景描述、小型无人机。AGX Orin 适合：复杂任务规划、VLM 视觉分析、多模型同时运行、中大型无人机。

**3.** 保障策略：(1) 模型预热 — 启动时预加载模型到内存；(2) 心跳检测 — 定期检查推理服务状态；(3) 超时处理 — 推理超时后使用默认响应；(4) 降级策略 — 推理失败时切换到规则引擎；(5) 看门狗 — 推理服务崩溃时自动重启。

**4.** 散热不足会导致：(1) GPU 降频 — 推理速度下降；(2) 系统不稳定 — 可能导致模型输出异常；(3) 寿命缩短 — 长期高温损坏硬件。解决方案：(1) 合理的散热设计；(2) 根据温度动态调整推理频率；(3) 监控温度，过热时暂停推理。

**5.** 完整方案：(1) 硬件选型 — 根据任务需求选择平台；(2) 系统安装 — JetPack + ROS2 + 依赖库；(3) 模型部署 — 量化模型 + 推理服务；(4) 系统集成 — ROS2 节点 + 通信 + 安全层；(5) 测试验证 — 仿真 → 悬停测试 → 飞行测试；(6) 运维 — 远程监控 + 日志 + 模型更新。
